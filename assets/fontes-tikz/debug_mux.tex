% Multiplexação da UART do USB pelo sinal RTS (soc_top.vhd).
\begin{tikzpicture}[font=\scriptsize,>={Stealth[length=1.6mm]},
  b/.style={draw,thick,align=center,minimum width=26mm,minimum height=10mm},
  sig/.style={font=\tiny\ttfamily}]
  \node[b,fill=gray!10,minimum height=30mm,minimum width=20mm] (host) at (0,0) {Computador\\(USB-UART)};
  % demux do RX
  \draw[thick,fill=gray!12] (3.0,1.55) -- (3.5,1.75) -- (3.5,0.35) -- (3.0,0.55) -- cycle;
  % mux do TX
  \draw[thick,fill=gray!12] (3.5,-0.55) -- (3.0,-0.35) -- (3.0,-1.75) -- (3.5,-1.55) -- cycle;
  \node[b,fill=blue!14] (soc) at (6.6,1.1) {UART do SoC\\{\tiny (programa do usuário)}};
  \node[b,fill=yellow!22] (dbg) at (6.6,-1.1) {Controlador\\de depuração};
  % RX
  \draw[->,thick] (1.0,1.05) -- node[sig,above]{RX} (3.0,1.05);
  \draw[->,thick] (3.5,1.35) -- node[sig,above]{RTS = 0} (5.3,1.35);
  \draw[->,thick] (3.5,0.75) -- (4.3,0.75) |- node[sig,pos=0.75,above]{RTS = 1} (5.3,-0.85);
  % TX
  \draw[->,thick] (5.3,0.85) -- (4.6,0.85) |- (3.5,-0.75);
  \draw[->,thick] (5.3,-1.35) -- (3.5,-1.35);
  \draw[->,thick] (3.0,-1.05) -- node[sig,above]{TX} (1.0,-1.05);
  % RTS
  \draw[thick,azul] (1.0,0) -- node[sig,above,text=azul]{RTS} (2.6,0);
  \draw[->,azul] (2.6,0) |- (3.25,0.3) -- (3.25,0.45);
  \draw[->,azul] (2.6,0) |- (3.25,-0.25) -- (3.25,-0.45);
  \node[font=\tiny,text=corpo,align=left,anchor=north west] at (1.0,-2.0) {o destino não selecionado recebe 1 (repouso da UART); o TX vem do\\controlador de depuração com RTS = 1 ou com a CPU parada};
\end{tikzpicture}
